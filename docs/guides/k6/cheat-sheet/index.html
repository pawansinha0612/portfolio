Commands and run flags

CommandDoes
k6 run script.jsRun a test (.js or .ts)
k6 inspect script.jsLoad the script and print the resolved options. No traffic. Good CI lint.
k6 new script.jsCreate a starter script
k6 archive script.jsBundle script, imports and files into a .tar
k6 cloud run script.jsRun in Grafana Cloud k6 (--local-execution runs here, streams results there)
k6 versionPrint the version. Check it first when an import fails.
k6 run flagDoes
-u, --vus 10Number of VUs
-d, --duration 5mDuration. With --vus: a constant-vus scenario.
-i, --iterations 100Total iterations, shared across VUs (shared-iterations)
-s, --stage 1m:50Add a stage (repeatable): ramp to 50 VUs over 1 m
-e, --env KEY=valueSet __ENV.KEY
-o, --out json=file.jsonSend metrics to an output (repeatable)
--tag testid=run42Tag every metric in the run
--http-debug / =fullLog request and response headers / and bodies
--summary-mode fullPer-group and per-scenario summary (v1.0+)
--summary-trend-stats "avg,p(95),p(99),max"Which stats the summary shows for trends
--no-thresholdsIgnore thresholds (debug only)
-q, --quietNo progress bar (CI logs)
--insecure-skip-tls-verifyAccept self-signed certificates

CLI flags like --vus, --duration and --iterations replace the script's scenarios. Precedence, highest first: CLI flags, then K6_* environment variables, then options in the script.

Executors

ExecutorModelKey settings
shared-iterationsClosedvus, iterations (total), maxDuration
per-vu-iterationsClosedvus, iterations (each), maxDuration
constant-vusClosedvus, duration
ramping-vusClosedstartVUs, stages[{duration, target}], gracefulRampDown
constant-arrival-rateOpenrate, timeUnit, duration, preAllocatedVUs, maxVUs
ramping-arrival-rateOpenstartRate, timeUnit, stages[{duration, target}], preAllocatedVUs, maxVUs

Every scenario also takes exec, startTime, gracefulStop (30 s), env, tags and options. VUs for arrival-rate: rate × iteration time (Little's Law). Missed starts go to dropped_iterations. externally-controlled was removed in v2.0.

Checks, thresholds, custom metrics

check(res, {
  'status 200': (r) => r.status === 200,
  'has token': (r) => r.json('access_token') !== undefined,
});                                    // returns true if all passed; never stops the test

thresholds: {
  http_req_duration: ['p(95)<500', 'p(99)<1500'],        // Trend: avg min max med p(N)
  http_req_failed: ['rate<0.01',                          // Rate: rate
    { threshold: 'rate<0.1', abortOnFail: true, delayAbortEval: '30s' }], // stop early
  http_reqs: ['count>1000'],                              // Counter: count, rate
  vus: ['value<=200'],                                    // Gauge: value
  checks: ['rate>0.99'],
  'http_req_duration{name:GET /products}': ['p(95)<300'], // by tag
  'http_req_duration{scenario:buy}': ['p(95)<800'],       // by scenario
  'group_duration{group:::checkout}': ['p(95)<2000'],     // group tag starts with ::
  'checks{check:status 200}': ['rate>0.999'],
}

import { Trend, Counter, Rate, Gauge } from 'k6/metrics';
const t = new Trend('checkout_time', true); t.add(ms);    // true = time values
const c = new Counter('orders');            c.add(1);
const r = new Rate('bad_body');             r.add(isBad);  // true/false
const g = new Gauge('queue_depth');         g.add(n);

Exit codes: 0 pass, 99 threshold failed, 104 invalid config, 107 script error, 108 exec.test.abort(). Note: a JavaScript object can't have the same key twice. Put several thresholds for one metric in one array.

Script skeleton

import http from 'k6/http';
import { check, group } from 'k6';
import { SharedArray } from 'k6/data';
import { Trend, Counter, Rate } from 'k6/metrics';
import exec from 'k6/execution';

const BASE = __ENV.BASE_URL || 'http://localhost:8080';
const users = new SharedArray('users', () => JSON.parse(open('./data/users.json')));
const cartTime = new Trend('cart_time', true);
const orders = new Counter('orders');
const badBody = new Rate('bad_body');

export const options = {
  scenarios: {
    load: {
      executor: 'ramping-arrival-rate',
      startRate: 0, timeUnit: '1s', preAllocatedVUs: 50, maxVUs: 300,
      stages: [{ duration: '5m', target: 100 }, { duration: '20m', target: 100 }, { duration: '2m', target: 0 }],
    },
  },
  thresholds: {
    http_req_failed: ['rate<0.01'],
    http_req_duration: ['p(95)<500', 'p(99)<1500'],
    'http_req_duration{name:POST /cart}': ['p(95)<800'],
    checks: ['rate>0.99'],
    dropped_iterations: ['count<10'],
  },
  discardResponseBodies: false,
  userAgent: 'k6-load-test',
};

export function setup() {
  const res = http.post(`${BASE}/login`, JSON.stringify({ username: 'svc', password: __ENV.PASSWORD }), {
    headers: { 'Content-Type': 'application/json' },
  });
  return { token: res.json('access_token'), startedAt: Date.now() };
}

export default function (data) {
  const user = users[exec.scenario.iterationInTest % users.length];
  group('cart', () => {
    const res = http.post(`${BASE}/cart`, JSON.stringify({ productId: 7, user: user.username }), {
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${data.token}` },
      tags: { name: 'POST /cart' },
    });
    const ok = check(res, { 'status 201': (r) => r.status === 201 });
    badBody.add(!ok);
    cartTime.add(res.timings.duration);
    if (ok) orders.add(1);
  });
  // no sleep() at the end: in an arrival-rate scenario the rate sets the pacing
}

export function teardown(data) {
  console.log(`ran for ${(Date.now() - data.startedAt) / 1000}s`);
}
